        let currentSlideIndex = 0;
        const slides = document.querySelectorAll('.banner-slide');
        const totalSlides = slides.length;
        let slideInterval = null;
        function showSlide(index) {
            currentSlideIndex = (index + totalSlides) % totalSlides;
            slides.forEach((slide, idx) => {
                if (idx === currentSlideIndex) {
                    slide.classList.remove('hidden');
                    slide.classList.add('block');
                } else {
                    slide.classList.remove('block');
                    slide.classList.add('hidden');
                }
            });
            const dot0 = document.getElementById('dot-0');
            const dot1 = document.getElementById('dot-1');
            if (dot0 && dot1) {
                if (currentSlideIndex === 0) {
                    dot0.className = 'w-6 h-2 rounded-full bg-mc-accent transition-all';
                    dot1.className = 'w-2.5 h-2 rounded-full bg-white/60 transition-all';
                } else {
                    dot1.className = 'w-6 h-2 rounded-full bg-mc-accent transition-all';
                    dot0.className = 'w-2.5 h-2 rounded-full bg-white/60 transition-all';
                }
            }
        }
        function changeSlide(step) {
            showSlide(currentSlideIndex + step);
            resetSlideTimer();
        }
        function goToSlide(index) {
            showSlide(index);
            resetSlideTimer();
        }
        function resetSlideTimer() {
            if (slideInterval) clearInterval(slideInterval);
            slideInterval = setInterval(() => {
                showSlide(currentSlideIndex + 1);
            }, 4800);
        }
        function toggleMobileMenu() {
            const drawer = document.getElementById('mobileMenuDrawer');
            if (drawer) {
                drawer.classList.toggle('hidden');
            }
        }
        function filterCategory(category) {
            const cards = document.querySelectorAll('.product-card');
            const buttons = document.querySelectorAll('[data-cat-btn]');
            buttons.forEach(btn => {
                if (btn.getAttribute('data-cat-btn') === category) {
                    btn.classList.add('active');
                } else {
                    btn.classList.remove('active');
                }
            });
            cards.forEach(card => {
                const itemCat = card.getAttribute('data-category');
                if (category === 'all' || itemCat === category) {
                    card.classList.remove('hidden');
                    card.classList.add('flex');
                } else {
                    card.classList.remove('flex');
                    card.classList.add('hidden');
                }
            });
        }

            e.preventDefault();

        /* Xóa toàn bộ padding/margin của tất cả thẻ cha WordPress bọc ngoài .mc-site-wrapper để triệt tiêu khoảng lệch phải 15px */
        function fixMobileRightShift() {
            const wrapper = document.querySelector('.mc-site-wrapper');
            if (!wrapper) return;
            let parent = wrapper.parentElement;
            while (parent && parent !== document.documentElement) {
                parent.style.setProperty('padding-left', '0px', 'important');
                parent.style.setProperty('padding-right', '0px', 'important');
                parent.style.setProperty('margin-left', '0px', 'important');
                parent.style.setProperty('margin-right', '0px', 'important');
                parent.style.setProperty('width', '100%', 'important');
                parent.style.setProperty('max-width', '100%', 'important');
                parent = parent.parentElement;
            }
            wrapper.style.setProperty('margin-left', '0px', 'important');
            wrapper.style.setProperty('width', '100%', 'important');
            wrapper.style.setProperty('max-width', '100%', 'important');
            const vw = document.documentElement.clientWidth || window.innerWidth;
            const rect = wrapper.getBoundingClientRect();
            if (Math.abs(rect.left) > 0.5 || rect.width > vw) {
                wrapper.style.setProperty('width', vw + 'px', 'important');
                wrapper.style.setProperty('max-width', vw + 'px', 'important');
                wrapper.style.setProperty('margin-left', (-rect.left) + 'px', 'important');
            }
        }
        fixMobileRightShift();
        window.addEventListener('DOMContentLoaded', fixMobileRightShift);
        window.addEventListener('load', function() {
            resetSlideTimer();
            fixMobileRightShift();
            setTimeout(fixMobileRightShift, 250);
        });
        window.addEventListener('resize', fixMobileRightShift);
